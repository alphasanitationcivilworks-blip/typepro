<script>


/* =========================
   TEXT DATA
========================= */

const paragraphs = [

"The quick brown fox jumps over the lazy dog. Learning to type faster takes regular practice and patience. Good typing skills help you work faster and become more productive every day.",

"Technology has changed the way people communicate, work, and learn. Good typing skills can save time every day. Practice regularly and focus on accuracy before trying to increase your typing speed.",

"Success does not come from doing something once. It comes from practicing consistently and learning from mistakes. Keep practicing every day and you will slowly improve your typing speed and accuracy.",

"Programming is a creative skill that allows people to build websites, applications, games, and useful digital tools. Learning new skills takes time, but consistent practice can make difficult things easier.",

"The best way to improve your typing speed is to focus on accuracy first and gradually increase your speed. Stay relaxed, keep your fingers positioned correctly, and avoid looking at the keyboard."

];


/* =========================
   ELEMENTS
========================= */

const textDisplay =
    document.getElementById("textDisplay");

const input =
    document.getElementById("input");

const timeDisplay =
    document.getElementById("time");

const wpmDisplay =
    document.getElementById("wpm");

const accuracyDisplay =
    document.getElementById("accuracy");

const mistakesDisplay =
    document.getElementById("mistakes");

const progress =
    document.getElementById("progress");

const startBtn =
    document.getElementById("startBtn");

const restartBtn =
    document.getElementById("restartBtn");

const result =
    document.getElementById("result");

const finalWpm =
    document.getElementById("finalWpm");

const finalAccuracy =
    document.getElementById("finalAccuracy");

const finalMistakes =
    document.getElementById("finalMistakes");

const finalTime =
    document.getElementById("finalTime");

const timeButtons =
    document.querySelectorAll(".time-btn");


/* =========================
   VARIABLES
========================= */

let currentText = "";

let timer = null;

let selectedTime = 180;

let timeLeft = 180;

let started = false;

let mistakes = 0;


/* =========================
   RANDOM TEXT
========================= */

function chooseText() {

    currentText =
        paragraphs[
            Math.floor(
                Math.random() *
                paragraphs.length
            )
        ];

    textDisplay.innerHTML = "";

    currentText
        .split("")
        .forEach((char, index) => {

            const span =
                document.createElement("span");

            span.textContent = char;

            if (index === 0) {

                span.classList.add(
                    "current"
                );

            }

            textDisplay.appendChild(span);

        });

}


/* =========================
   TIMER FORMAT
========================= */

function updateTimer() {

    const minutes =
        Math.floor(timeLeft / 60);

    const seconds =
        timeLeft % 60;

    timeDisplay.textContent =
        minutes +
        ":" +
        String(seconds).padStart(2, "0");

}


/* =========================
   SELECT TIME
========================= */

timeButtons.forEach(button => {

    button.addEventListener(
        "click",
        () => {

            if (started) return;

            timeButtons.forEach(btn => {

                btn.classList.remove(
                    "active"
                );

            });

            button.classList.add(
                "active"
            );

            selectedTime =
                Number(
                    button.dataset.time
                );

            timeLeft =
                selectedTime;

            updateTimer();

        }
    );

});


/* =========================
   START
========================= */

function startTest() {

    if (started) return;

    started = true;

    input.disabled = false;

    input.focus();

    startBtn.disabled = true;

    timer =
        setInterval(() => {

            timeLeft--;

            updateTimer();

            calculateStats();

            if (timeLeft <= 0) {

                finishTest();

            }

        }, 1000);

}


/* =========================
   CALCULATE
========================= */

function calculateStats() {

    const typedText =
        input.value;

    const spans =
        textDisplay.querySelectorAll(
            "span"
        );

    mistakes = 0;


    spans.forEach(
        (span, index) => {

            span.classList.remove(
                "correct",
                "incorrect",
                "current"
            );


            if (
                index <
                typedText.length
            ) {

                if (
                    typedText[index] ===
                    currentText[index]
                ) {

                    span.classList.add(
                        "correct"
                    );

                } else {

                    span.classList.add(
                        "incorrect"
                    );

                    mistakes++;

                }

            }

            else if (
                index ===
                typedText.length
            ) {

                span.classList.add(
                    "current"
                );

            }

        }
    );


    const correctCharacters =
        Math.max(
            0,
            typedText.length -
            mistakes
        );


    const elapsedSeconds =
        selectedTime -
        timeLeft;


    const elapsedMinutes =
        elapsedSeconds / 60;


    let wpm = 0;


    if (elapsedMinutes > 0) {

        wpm =
            Math.round(
                (
                    correctCharacters /
                    5
                ) /
                elapsedMinutes
            );

    }


    const accuracy =
        typedText.length > 0

        ? Math.round(
            (
                correctCharacters /
                typedText.length
            ) * 100
        )

        : 100;


    wpmDisplay.textContent =
        Math.max(0, wpm);

    accuracyDisplay.textContent =
        accuracy + "%";

    mistakesDisplay.textContent =
        mistakes;


    /* Progress */

    const percentage =
        Math.min(
            100,
            (
                typedText.length /
                currentText.length
            ) * 100
        );

    progress.style.width =
        percentage + "%";


    /* Complete text */

    if (
        typedText.length >=
        currentText.length
    ) {

        finishTest();

    }

}


/* =========================
   FINISH
========================= */

function finishTest() {

    clearInterval(timer);

    input.disabled = true;

    calculateStats();


    finalWpm.textContent =
        wpmDisplay.textContent;

    finalAccuracy.textContent =
        accuracyDisplay.textContent;

    finalMistakes.textContent =
        mistakes;

    finalTime.textContent =
        formatTime(
            selectedTime -
            timeLeft
        );


    result.style.display =
        "block";


    started = false;

    startBtn.disabled =
        false;

}


/* =========================
   TIME FORMAT
========================= */

function formatTime(seconds) {

    const minutes =
        Math.floor(seconds / 60);

    const secs =
        seconds % 60;

    return (
        minutes +
        ":" +
        String(secs).padStart(2, "0")
    );

}


/* =========================
   RESET
========================= */

function resetTest() {

    clearInterval(timer);

    timeLeft =
        selectedTime;

    mistakes = 0;

    started = false;

    input.value = "";

    input.disabled = true;

    updateTimer();

    wpmDisplay.textContent =
        "0";

    accuracyDisplay.textContent =
        "100%";

    mistakesDisplay.textContent =
        "0";

    progress.style.width =
        "0%";

    result.style.display =
        "none";

    startBtn.disabled =
        false;

    chooseText();

}


/* =========================
   EVENTS
========================= */

startBtn.addEventListener(
    "click",
    startTest
);

restartBtn.addEventListener(
    "click",
    resetTest
);

input.addEventListener(
    "input",
    calculateStats
);


/* =========================
   INITIAL
========================= */

chooseText();

updateTimer();

</script>
